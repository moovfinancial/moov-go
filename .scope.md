// ImageConflictError represents a 409 Conflict response when uploading a duplicate image.
// Contains the metadata of the existing image that matches the uploaded content.
// Available when using API version v2026.10.00 or later.
type ImageConflictError struct {
    ExistingImage *ImageMetadata
}

// Error implements the error interface.
func (e *ImageConflictError) Error() string {
    if e.ExistingImage != nil {
        return fmt.Sprintf("image already exists with ID: %s", e.ExistingImage.ImageID)
    }
    return "image already exists"
}

// UploadImageV2026_10 uploads a new PNG, JPEG, or WebP image with optional metadata using API version v2026.10.00.
// If the image already exists (duplicate by content hash), returns *ImageConflictError containing the existing image's metadata.
// https://docs.moov.io/api/tools/images/post/
func (c Client) UploadImageV2026_10(ctx context.Context, accountID string, file io.Reader, metadata *ImageMetadataRequest) (*ImageMetadata, error) {
    var multiParts []multipartFn
    multiParts = append(multiParts, MultipartFile("image", "image", file, "application/octet-stream"))

    if metadata != nil {
        mdJson, err := json.Marshal(metadata)
        if err != nil {
            return nil, err
        }
        multiParts = append(multiParts, MultipartField("metadata", string(mdJson)))
    }

    resp, err := c.CallHttp(ctx, Endpoint(http.MethodPost, pathImages, accountID), MoovVersion(Version2026_10), MultipartBody(multiParts...))
    if err != nil {
        return nil, err
    }

    switch resp.Status() {
    case StatusStarted:
        return UnmarshalObjectResponse[ImageMetadata](resp)
    case StatusStateConflict:
        existing, unmarshalErr := UnmarshalObjectResponse[ImageMetadata](resp)
        if unmarshalErr == nil && existing != nil {
            return nil, &ImageConflictError{ExistingImage: existing}
        }
        return nil, resp
    default:
        return nil, resp
    }
}
